#include <iostream>
#include <thread>
#include <chrono>
#include <cstdlib>
#include <ctime>
using namespace std;

void wait(int ms) {
    this_thread::sleep_for(chrono::milliseconds(ms));
}

void clearScreen() {
    cout << "\033[2J\033[1;1H";
}

// Draw a big heart
void bigHeart(bool big) {
    if (big) {
        cout << R"(
                 ***       ***
              ********* *********
            ***********************
           *************************
          ***************************
          ***************************
           *************************
            ***********************
             *********************
               *****************
                 *************
                   *********
                     *****
                       *
)";
    } else {
        cout << R"(
               **     **
             ******* *******
           *******************
          *********************
          *********************
           *******************
             ***************
               ***********
                 *******
                   ***
                    *
)";
    }
}

// Falling stars
void stars(int frame) {
    string spaces[] = {
        "     ", "          ", "  ", "              ",
        "        ", "    ", "           ", " "
    };

    cout << "\n";

    for (int i = 0; i < 8; i++) {
        cout << spaces[(i + frame) % 8]
             << ((i + frame) % 2 ? "*" : ".")
             << "\n";
    }
}

    string theirName = "THEIR NAME";
